# 🚀 BuilderConnect Enterprise Expansion Plan

## **Executive Summary**

This comprehensive expansion transforms BuilderConnect from a basic marketplace into a **enterprise-grade construction platform** comparable to BookaBuilderUK, ConstructionMarket.io, and ConstructConnect. The plan introduces advanced construction-industry features, professional SaaS UI/UX, modular architecture, and scalable infrastructure while maintaining technical feasibility for Pakistan's construction ecosystem.

**Date:** January 13, 2026
**Authors:** Senior Software Architect, Product Manager, UI/UX Lead
**Target Platforms:** BookaBuilderUK, ConstructionMarket.io, ConstructConnect

## **5️⃣ Database & Domain Model Expansion**

### **New Core Entities**

#### **Project Lifecycle Entities**
```sql
-- Enhanced Projects
CREATE TABLE projects (
    id BIGINT PRIMARY KEY,
    client_id BIGINT REFERENCES users(id),
    title VARCHAR(255) NOT NULL,
    description TEXT,
    category VARCHAR(50),
    trade_type VARCHAR(50),
    budget_min DECIMAL(12,2),
    budget_max DECIMAL(12,2),
    location_lat DECIMAL(10,8),
    location_lng DECIMAL(11,8),
    city VARCHAR(100),
    status ENUM('DRAFT', 'OPEN', 'AWARDED', 'IN_PROGRESS', 'COMPLETED', 'CANCELLED'),
    priority ENUM('LOW', 'MEDIUM', 'HIGH', 'URGENT'),
    created_at TIMESTAMP,
    updated_at TIMESTAMP
);

-- Project Milestones
CREATE TABLE project_milestones (
    id BIGINT PRIMARY KEY,
    project_id BIGINT REFERENCES projects(id),
    title VARCHAR(255),
    description TEXT,
    sequence_number INT,
    estimated_days INT,
    payment_percentage DECIMAL(5,2),
    status ENUM('PENDING', 'IN_PROGRESS', 'COMPLETED', 'DELAYED'),
    actual_completion_date DATE
);

-- Contracts
CREATE TABLE contracts (
    id BIGINT PRIMARY KEY,
    project_id BIGINT REFERENCES projects(id),
    builder_id BIGINT REFERENCES users(id),
    contract_template_id BIGINT,
    total_amount DECIMAL(12,2),
    signed_date DATE,
    status ENUM('DRAFT', 'SIGNED', 'ACTIVE', 'COMPLETED', 'TERMINATED'),
    pdf_url VARCHAR(500)
);
```

#### **Financial Management**
```sql
-- Payments & Escrow
CREATE TABLE payments (
    id BIGINT PRIMARY KEY,
    project_id BIGINT REFERENCES projects(id),
    milestone_id BIGINT REFERENCES project_milestones(id),
    payer_id BIGINT REFERENCES users(id),
    payee_id BIGINT REFERENCES users(id),
    amount DECIMAL(12,2),
    status ENUM('PENDING', 'HELD_IN_ESCROW', 'RELEASED', 'REFUNDED'),
    payment_method VARCHAR(50),
    transaction_id VARCHAR(255),
    released_at TIMESTAMP
);

-- Subscriptions
CREATE TABLE subscriptions (
    id BIGINT PRIMARY KEY,
    user_id BIGINT REFERENCES users(id),
    plan_type ENUM('FREE', 'BASIC', 'PRO', 'ENTERPRISE'),
    status ENUM('ACTIVE', 'PAUSED', 'CANCELLED', 'EXPIRED'),
    current_period_start DATE,
    current_period_end DATE,
    cancel_at_period_end BOOLEAN,
    lead_credits_remaining INT
);
```

#### **Builder Enhancement Entities**
```sql
-- Company Profiles
CREATE TABLE company_profiles (
    id BIGINT PRIMARY KEY,
    builder_id BIGINT REFERENCES users(id),
    company_name VARCHAR(255),
    ntn_number VARCHAR(50),
    pec_registration VARCHAR(50),
    established_year YEAR,
    employee_count INT,
    service_radius_km INT,
    website_url VARCHAR(255),
    description TEXT
);

-- Trade Specializations
CREATE TABLE builder_trades (
    id BIGINT PRIMARY KEY,
    builder_id BIGINT REFERENCES users(id),
    trade_type VARCHAR(50),
    experience_years INT,
    certification_level VARCHAR(50),
    primary_trade BOOLEAN DEFAULT FALSE
);

-- Certifications & Licenses
CREATE TABLE certifications (
    id BIGINT PRIMARY KEY,
    builder_id BIGINT REFERENCES users(id),
    certification_type VARCHAR(100),
    issuing_authority VARCHAR(100),
    certificate_number VARCHAR(100),
    issue_date DATE,
    expiry_date DATE,
    document_url VARCHAR(500),
    verification_status ENUM('PENDING', 'VERIFIED', 'REJECTED')
);
```

#### **Platform Management**
```sql
-- Disputes & Resolution
CREATE TABLE disputes (
    id BIGINT PRIMARY KEY,
    project_id BIGINT REFERENCES projects(id),
    initiator_id BIGINT REFERENCES users(id),
    respondent_id BIGINT REFERENCES users(id),
    dispute_type ENUM('PAYMENT', 'QUALITY', 'TIMELINE', 'CONTRACT_BREACH'),
    status ENUM('OPEN', 'UNDER_REVIEW', 'RESOLVED', 'ESCALATED'),
    resolution TEXT,
    resolved_by BIGINT REFERENCES users(id),
    created_at TIMESTAMP
);

-- Notifications
CREATE TABLE notifications (
    id BIGINT PRIMARY KEY,
    user_id BIGINT REFERENCES users(id),
    type ENUM('PROJECT_UPDATE', 'BID_RECEIVED', 'PAYMENT_DUE', 'REVIEW_REQUEST'),
    title VARCHAR(255),
    message TEXT,
    read_status BOOLEAN DEFAULT FALSE,
    action_url VARCHAR(500),
    created_at TIMESTAMP
);

-- Activity Logs
CREATE TABLE activity_logs (
    id BIGINT PRIMARY KEY,
    user_id BIGINT REFERENCES users(id),
    action_type VARCHAR(100),
    entity_type VARCHAR(50),
    entity_id BIGINT,
    old_values JSON,
    new_values JSON,
    ip_address VARCHAR(45),
    user_agent TEXT,
    created_at TIMESTAMP
);
```
